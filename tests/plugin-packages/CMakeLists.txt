cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NotepadNextPluginPackageTests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    find_package(Qt6 REQUIRED COMPONENTS Core Network Gui Test)
    qt_standard_project_setup()
    enable_testing()
    add_subdirectory(../../src/plugins/sources plugin-sources)
    add_subdirectory(../../src/plugins/packages plugin-packages)
endif()
include(${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/PluginZlib.cmake)
add_executable(PluginPackageTests PluginPackageTests.cpp)
add_executable(PackageLeaseProbe PackageLeaseProbe.cpp)
target_link_libraries(PackageLeaseProbe PRIVATE NotepadNextPluginPackages)
add_dependencies(PluginPackageTests PackageLeaseProbe)
target_link_libraries(PluginPackageTests PRIVATE NotepadNextPluginPackages Qt6::Test ZLIB::ZLIB)
add_test(NAME PluginPackageTests COMMAND PluginPackageTests -o -,txt -o "${CMAKE_BINARY_DIR}/PluginPackageTests.qt.txt,txt")
set_tests_properties(PluginPackageTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(PackageTransactionProbe PackageTransactionProbe.cpp)
target_link_libraries(PackageTransactionProbe PRIVATE NotepadNextPluginPackages)
add_executable(PluginTransactionCrashTests TransactionCrashTests.cpp)
target_link_libraries(PluginTransactionCrashTests PRIVATE NotepadNextPluginPackages Qt6::Test)
add_dependencies(PluginTransactionCrashTests PackageTransactionProbe)
add_test(NAME PluginTransactionCrashTests COMMAND PluginTransactionCrashTests -o -,txt -o "${CMAKE_BINARY_DIR}/PluginTransactionCrashTests.qt.txt,txt")
set_tests_properties(PluginTransactionCrashTests PROPERTIES TIMEOUT 60)
add_executable(PluginGitHubReleaseTests GitHubReleaseTests.cpp)
target_link_libraries(PluginGitHubReleaseTests PRIVATE NotepadNextPluginPackages Qt6::Test ZLIB::ZLIB)
if(WIN32)
    target_link_libraries(PluginGitHubReleaseTests PRIVATE advapi32)
endif()
add_test(NAME PluginGitHubReleaseTests COMMAND PluginGitHubReleaseTests -o -,txt -o "${CMAKE_BINARY_DIR}/PluginGitHubReleaseTests.qt.txt,txt")
set(NOTEPADNEXT_TEST_NODE "" CACHE FILEPATH "Absolute Node 24 executable for integration tests")
if(NOTEPADNEXT_TEST_NODE AND UNIX)
    if(NOT TARGET NotepadNextPluginRpc)
        add_subdirectory(../../src/plugins/rpc plugin-rpc)
    endif()
    add_executable(ParentCrashProbe ParentCrashProbe.cpp)
    target_link_libraries(ParentCrashProbe PRIVATE NotepadNextPluginPackages NotepadNextPluginRpc)
    add_executable(PluginParentCrashTests ParentCrashTests.cpp)
    target_link_libraries(PluginParentCrashTests PRIVATE NotepadNextPluginPackages Qt6::Test)
    target_compile_definitions(PluginParentCrashTests PRIVATE PARENT_TEST_NODE="${NOTEPADNEXT_TEST_NODE}" PARENT_TEST_HOST="${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-host/host.cjs" PARENT_TEST_PYTHON="${NOTEPADNEXT_TEST_PYTHON}")
    add_dependencies(PluginParentCrashTests ParentCrashProbe)
    add_test(NAME PluginParentCrashTests COMMAND PluginParentCrashTests -o -,txt -o "${CMAKE_BINARY_DIR}/PluginParentCrashTests.qt.txt,txt")
    set_tests_properties(PluginParentCrashTests PROPERTIES TIMEOUT 20)
endif()
