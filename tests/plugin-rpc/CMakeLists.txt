cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NotepadNextPluginRpcTests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    find_package(Qt6 REQUIRED COMPONENTS Core Test)
    qt_standard_project_setup()
    enable_testing()
    add_subdirectory(../../src/plugins/rpc plugin-rpc)
endif()
set(NOTEPADNEXT_TEST_PYTHON "" CACHE FILEPATH "Absolute Python 3.12 executable for second-language integration tests")
if(NOTEPADNEXT_TEST_PYTHON)
    add_test(NAME PluginPythonContractTests COMMAND "${NOTEPADNEXT_TEST_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/python.test.py")
    if(NOTEPADNEXT_TEST_NODE)
        add_test(NAME PluginPythonHostTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/python-host.test.cjs")
        set_tests_properties(PluginPythonHostTests PROPERTIES ENVIRONMENT "NNP_TEST_PYTHON=${NOTEPADNEXT_TEST_PYTHON}")
    endif()
endif()
set(NOTEPADNEXT_TEST_NODE "" CACHE FILEPATH "Absolute Node 24 executable for integration tests")
add_executable(PluginRpcTests PluginRpcTests.cpp)
target_link_libraries(PluginRpcTests PRIVATE NotepadNextPluginRpc Qt6::Test)
target_compile_definitions(PluginRpcTests PRIVATE
    RPC_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    RPC_TEST_NODE="${NOTEPADNEXT_TEST_NODE}"
)
add_test(NAME PluginRpcTests COMMAND PluginRpcTests)
if(NOTEPADNEXT_TEST_NODE)
    add_test(NAME PluginNodeRpcTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/rpc.test.cjs")
    add_test(NAME PluginNodeHostTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/host.test.cjs")
    add_test(NAME PluginNodeWireTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/wire.test.cjs")
    add_test(NAME PluginDocumentMirrorTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/mirror.test.cjs")
    add_test(NAME PluginVSCodeCoreTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/vscode.test.cjs")
    add_test(NAME PluginVSCodeWebviewsTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/webviews.test.cjs")
    add_test(NAME PluginVSCodeSampleBuild COMMAND "${NOTEPADNEXT_TEST_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/vscode-hello-world/build.cjs")
    add_test(NAME PluginVSCodeWebviewSampleBuild COMMAND "${NOTEPADNEXT_TEST_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/vscode-webview/build.cjs")
    add_test(NAME PluginSdkPackageTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-sdk/pack.test.cjs")
    add_test(NAME PluginSdkDeveloperTests COMMAND "${NOTEPADNEXT_TEST_NODE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-sdk/developer.test.cjs" "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-sdk/local-service.test.cjs")
    add_test(NAME PluginPythonSampleBuild COMMAND "${NOTEPADNEXT_TEST_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/python-inspector/build.cjs")
    set_tests_properties(PluginPythonSampleBuild PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/python-inspector")
