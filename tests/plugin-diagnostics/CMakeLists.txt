cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NotepadNextPluginDiagnosticsTests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    find_package(Qt6 REQUIRED COMPONENTS Core Test)
    qt_standard_project_setup()
    enable_testing()
endif()
if(NOT TARGET NotepadNextPluginDiagnostics)
    add_subdirectory(../../src/plugins/diagnostics plugin-diagnostics)
endif()
add_executable(PluginDiagnosticTests PluginDiagnosticTests.cpp)
add_executable(DiagnosticCrashProbe DiagnosticCrashProbe.cpp)
target_link_libraries(DiagnosticCrashProbe PRIVATE Qt6::Core)
add_dependencies(PluginDiagnosticTests DiagnosticCrashProbe)
target_link_libraries(PluginDiagnosticTests PRIVATE NotepadNextPluginDiagnostics Qt6::Test)
add_test(NAME PluginDiagnosticTests COMMAND PluginDiagnosticTests)
set_tests_properties(PluginDiagnosticTests PROPERTIES TIMEOUT 20)
