if(NOT TARGET NotepadNextAppearance)
    add_subdirectory(../../appearance ${CMAKE_CURRENT_BINARY_DIR}/appearance)
endif()
if(NOT TARGET NotepadNextPluginPackages)
    find_package(Qt6 REQUIRED COMPONENTS Network Gui)
    add_subdirectory(../sources ${CMAKE_CURRENT_BINARY_DIR}/plugin-sources)
    add_subdirectory(../packages ${CMAKE_CURRENT_BINARY_DIR}/plugin-packages)
endif()
if(NOT TARGET NotepadNextPluginData)
    add_subdirectory(../data ${CMAKE_CURRENT_BINARY_DIR}/plugin-data)
endif()
if(NOT TARGET NotepadNextPluginDiagnostics)
    add_subdirectory(../diagnostics ${CMAKE_CURRENT_BINARY_DIR}/plugin-diagnostics)
endif()
add_library(NotepadNextPluginRuntime STATIC PluginSession.cpp PluginSession.h PluginEngine.cpp PluginEngine.h)
target_link_libraries(NotepadNextPluginRuntime PUBLIC NotepadNextAppearance NotepadNextPluginDocuments NotepadNextPluginPackages NotepadNextPluginData NotepadNextPluginDiagnostics)
target_include_directories(NotepadNextPluginRuntime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(NotepadNextPluginRuntime PROPERTIES AUTOMOC ON)
set(NOTEPADNEXT_PLUGIN_PYTHON "" CACHE FILEPATH "Explicit Python 3.12 interpreter for developer prototype plugins; not bundled")
if(NOTEPADNEXT_PLUGIN_PYTHON)
    if(NOT IS_ABSOLUTE "${NOTEPADNEXT_PLUGIN_PYTHON}" OR NOT EXISTS "${NOTEPADNEXT_PLUGIN_PYTHON}")
        message(FATAL_ERROR "NOTEPADNEXT_PLUGIN_PYTHON must be an existing absolute interpreter")
    endif()
    execute_process(COMMAND "${NOTEPADNEXT_PLUGIN_PYTHON}" -I -c "import sys; print('.'.join(map(str,sys.version_info[:2])))"
        RESULT_VARIABLE plugin_python_status OUTPUT_VARIABLE plugin_python_version OUTPUT_STRIP_TRAILING_WHITESPACE TIMEOUT 5)
    if(NOT plugin_python_status EQUAL 0 OR NOT plugin_python_version STREQUAL "3.12")
        message(FATAL_ERROR "The developer Python prototype requires Python 3.12")
    endif()
    target_compile_definitions(NotepadNextPluginRuntime PRIVATE NOTEPADNEXT_PLUGIN_PYTHON_PATH="${NOTEPADNEXT_PLUGIN_PYTHON}")
endif()
if(TARGET NotepadNextPluginWebviews)
    target_link_libraries(NotepadNextPluginRuntime PUBLIC NotepadNextPluginWebviews)
    target_compile_definitions(NotepadNextPluginRuntime PUBLIC NOTEPADNEXT_PLUGIN_WEBVIEWS)
endif()
